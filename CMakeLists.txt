option(RR_BUILD_LEGACY_IOP_CHECKS "Compare idle adapter with the preserved native IOP" OFF)
if(RR_BUILD_LEGACY_IOP_CHECKS)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "Legacy reference libraries require Windows x64")
    endif()
    set(legacy_iop "${CMAKE_SOURCE_DIR}/work/native/sources/src/PS2Recomp/ps2xIOP")
    foreach(variant IN ITEMS idle reference)
        if(variant STREQUAL "idle")
            set(adapter "${CMAKE_SOURCE_DIR}/build/legacy-iop47B2/iop_emulator.cpp")
        else()
            set(adapter "${legacy_iop}/src/emulator/iop_emulator.cpp")
        endif()
        add_executable(racer_legacy_iop_${variant}_check tests/legacy_iop_idle47b.cpp ${adapter})
        target_compile_features(racer_legacy_iop_${variant}_check PRIVATE cxx_std_20)
        target_include_directories(racer_legacy_iop_${variant}_check PRIVATE src/core
            "${legacy_iop}/include" "${legacy_iop}/src" "${legacy_iop}/src/emulator" "${legacy_iop}/tests")
        target_link_libraries(racer_legacy_iop_${variant}_check PRIVATE "${CMAKE_SOURCE_DIR}/work/diagnostic45y/ps2_iop.lib")
    endforeach()
endif()

        message(FATAL_ERROR "Xbox object validation needs the Windows x86 target")

    set(xbox_objects "${CMAKE_SOURCE_DIR}/src/core/cycle_ledger.obj" "${CMAKE_SOURCE_DIR}/src/core/compact_registry.obj" "${CMAKE_SOURCE_DIR}/tests/core_suite.obj")
